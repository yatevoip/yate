cmake_minimum_required(VERSION 3.16)
project(Yate VERSION 6.4.1 LANGUAGES C CXX)

if(WIN32 OR NOT UNIX)
    message(FATAL_ERROR "This fork supports Unix systems only")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build: cmake -S . -B build")
endif()

include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()

option(YATE_BUILD_MODULES "Build loadable telephony modules" ON)
option(YATE_BUILD_TEST_MODULES "Build the original test plugins and SS7 utility" ${BUILD_TESTING})
option(YATE_STATIC_STRINGS "Use compiler-supported static String expressions" ON)
option(YATE_ATOMICS "Use compiler-supported atomic operations" ON)
option(YATE_POLL "Use poll() rather than select() when available" ON)
option(YATE_SSE2 "Enable SSE2 in bundled miniwebrtc" OFF)
option(YATE_ISAC_FIXED "Use fixed-point rather than floating-point iSAC" OFF)
set(YATE_MODULES all CACHE STRING "Modules to build: all, none, or a semicolon-separated list of names")
set(YATE_RTTI AUTO CACHE STRING "RTTI: AUTO, ON or OFF")
set_property(CACHE YATE_RTTI PROPERTY STRINGS AUTO ON OFF)
set(YATE_FDSIZE 8192 CACHE STRING "select() descriptor limit on glibc; 0 disables the override")
set(YATE_RELEASE 1 CACHE STRING "Package release")
set(YATE_STATUS devel CACHE STRING "Package status")
set(YATE_INSTALL_MODULEDIR "${CMAKE_INSTALL_LIBDIR}/yate" CACHE PATH "Plugin installation directory")
set(YATE_INSTALL_CONFIGDIR "${CMAKE_INSTALL_SYSCONFDIR}/yate" CACHE PATH "Configuration installation directory")
set(YATE_INSTALL_SHAREDIR "${CMAKE_INSTALL_DATADIR}/yate" CACHE PATH "Resource installation directory")

include(YateHelpers)
include(PlatformChecks)
include(Dependencies)

set(PACKAGE_VERSION "${PROJECT_VERSION}")
set(PACKAGE_VERSION_MAJOR "${PROJECT_VERSION_MAJOR}")
set(PACKAGE_VERSION_MINOR "${PROJECT_VERSION_MINOR}")
set(PACKAGE_VERSION_BUILD "${PROJECT_VERSION_PATCH}")
set(PACKAGE_RELEASE "${YATE_RELEASE}")
set(PACKAGE_STATUS "${YATE_STATUS}")
set(PACKAGE_REVISION unknown)
find_package(Git QUIET)
if(Git_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PROJECT_SOURCE_DIR}" rev-parse HEAD
        OUTPUT_VARIABLE _revision OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(_revision)
        set(PACKAGE_REVISION "${_revision}")
    endif()
endif()
if(PACKAGE_REVISION STREQUAL "unknown")
    foreach(_revision_file packing/revision.txt packing/git_commit.txt)
        if(EXISTS "${PROJECT_SOURCE_DIR}/${_revision_file}")
            file(STRINGS "${PROJECT_SOURCE_DIR}/${_revision_file}" PACKAGE_REVISION LIMIT_COUNT 1)
            break()
        endif()
    endforeach()
endif()
set(PACKAGE_GIT_HASH "${PACKAGE_REVISION}")
configure_file(yateversn.h.in yateversn.h @ONLY)
foreach(_dir MODULE CONFIG SHARE)
    yate_absolute_install_path(YATE_FULL_${_dir}DIR "${YATE_INSTALL_${_dir}DIR}")
endforeach()
configure_file(cmake/yatepaths.h.in yatepaths.h @ONLY)

add_subdirectory(engine)
if(YATE_BUILD_MODULES OR YATE_BUILD_TEST_MODULES)
    add_subdirectory(libs)
endif()
if(YATE_BUILD_MODULES)
    add_subdirectory(modules)
endif()
add_executable(yate main.cpp)
target_link_libraries(yate PRIVATE libyate yate_build_options)
yate_install_rpath(yate "${CMAKE_INSTALL_BINDIR}")
install(TARGETS yate RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")

include(InstallYate)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

get_property(_built GLOBAL PROPERTY YATE_ENABLED_MODULES)
get_property(_skipped GLOBAL PROPERTY YATE_DISABLED_MODULES)
list(LENGTH _built _module_count)
message(STATUS "Yate ${PROJECT_VERSION}: ${_module_count} modules")
message(STATUS "Enabled modules: ${_built}")
message(STATUS "Unavailable optional modules: ${_skipped}")
message(STATUS "Config: ${YATE_FULL_CONFIGDIR}; plugins: ${YATE_FULL_MODULEDIR}")
